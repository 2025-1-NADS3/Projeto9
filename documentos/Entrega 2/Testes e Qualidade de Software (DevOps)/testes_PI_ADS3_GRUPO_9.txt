
import sqlite3

# Classe responsável por simular o banco de dados de usuários
class UserDatabase:
    def __init__(self):
        # Banco de dados em memória (não salva em arquivo)
        self.conn = sqlite3.connect(':memory:')
        self.create_table()

    # Cria a tabela 'Users' no banco de dados
    def create_table(self):
        self.conn.execute('''
            CREATE TABLE Users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                firstName TEXT,
                lastName TEXT,
                email TEXT,
                ra TEXT,
                password TEXT
            )
        ''')
        self.conn.commit()

    # Insere um novo usuário na tabela
    def insert_user(self, firstName, lastName, email, ra, password):
        self.conn.execute('''
            INSERT INTO Users (firstName, lastName, email, ra, password)
            VALUES (?, ?, ?, ?, ?)
        ''', (firstName, lastName, email, ra, password))
        self.conn.commit()

    # Realiza login com RA e senha, retorna o usuário se encontrar
    def login_user(self, ra, password):
        cursor = self.conn.cursor()
        cursor.execute('''
            SELECT * FROM Users WHERE ra = ? AND password = ?
        ''', (ra, password))
        return cursor.fetchone()


# Teste 1 – Verifica inserção e login de usuário válido
def teste1():
    print("\n Teste 1 – Inserção de Usuário com Dados Válidos")
    db = UserDatabase()
    db.insert_user("João", "Pereira", "joao@email.com", "123456", "senha123")
    user = db.login_user("123456", "senha123")

    if user and user[1] == "João":
        print("SUCESSO: Usuário João inserido e encontrado.")
    else:
        print("FALHA: Usuário João não foi localizado.")
